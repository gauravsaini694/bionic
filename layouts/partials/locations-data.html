{{- /* Single source of truth: builds states + clinics from content/locations/ (states) and content/clinics/ (clinics). Returns a dict. */ -}} {{- $loc := site.GetPage "/locations" -}} {{- $states := slice -}} {{- $clinics := slice -}} {{- range $loc.Sections.ByTitle -}} {{- $st := . -}} {{- $n := 0 -}} {{- range (where site.RegularPages "Type" "clinic").ByTitle -}}{{- if eq .Params.state $st.Title -}} {{- $zip := .Params.zip -}} {{- if not $zip -}} {{- with findRE `\b\d{5}\s*$` (partial "clinic-address.html" . | strings.TrimSpace) 1 -}}{{- $zip = strings.TrimSpace (index . 0) -}}{{- end -}} {{- end -}} {{- $c := dict "name" .Title "city" (.Params.city | default .Title) "state" $st.Title "stateUrl" $st.RelPermalink "address" (partial "clinic-address.html" .) "phone" (.Params.phone | default "") "zip" ($zip | default "") "url" .RelPermalink -}} {{- with .Params.lat }}{{ $c = merge $c (dict "lat" .) }}{{ end -}} {{- with .Params.lng }}{{ $c = merge $c (dict "lng" .) }}{{ end -}} {{- $clinics = $clinics | append $c -}} {{- $n = add $n 1 -}} {{- end -}} {{- end -}} {{- if $n -}} {{- $states = $states | append (dict "name" $st.Title "url" $st.RelPermalink "count" $n) -}} {{- end -}} {{- end -}} {{- return (dict "states" $states "clinics" $clinics "total" (len $clinics) "stateCount" (len $states)) -}}

{{- /* Escaped text with inline links. Syntax: **bold**, trailing " :: description" (small second line), [[label|/path/]] or [[label|https://x|_blank]] (_blank => new tab). Context: string. */ -}} {{- $s := . | string | htmlEscape -}} {{- $s = replaceRE `\*\*([^*]+)\*\*` `${1}` $s -}} {{- $s = replaceRE `\[\[([^\[\]|]+)\|([^\[\]|]+)\|_blank\]\]` `${1}` $s -}} {{- $s = replaceRE `\[\[([^\[\]|]+)\|([^\[\]|]+)\]\]` `${1}` $s -}} {{- $s = replaceRE ` :: (.+)$` `${1}` $s -}} {{- return ($s | safeHTML) -}}
